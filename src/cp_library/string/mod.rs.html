6/// Calculate [Z-function](https://cp-algorithms.com/string/z-function.html)
7/// of a given slice of an [`Eq`] type `T`.
8///
9/// Complexity: $\mathcal{O}(N)$ comparisons where:
10/// - $N$ is the size of the slice.
11///
12/// # Examples
13///
14/// ```
15/// use cp_library::string::z_array;
16///
17/// let arr = "anananasso";
18/// let z = z_array(arr.as_bytes());
19///
20/// assert_eq!(z, [10, 0, 5, 0, 3, 0, 1, 0, 0, 0]);
21/// ```
22pub fn z_array<T: Eq>(arr: &[T]) -> Vec<usize> {
23    let mut z = vec![0; arr.len()];
24    let mut l = 0;
25    let mut r = 0;
26
27    z[0] = arr.len();
28    for i in 1..arr.len() {
29        if i <= r {
30            z[i] = cmp::min(r - i, z[i - l]);
31        }
32        while i + z[i] < z.len() && arr[i + z[i]] == arr[z[i]] {
33            z[i] += 1;
34        }
35        if i + z[i] > r {
36            l = i;
37            r = i + z[i];
38        }
39    }
40
41    z
42}

44/// Cyclic dictionary basic factor of a string over an alphabet `T`.
45///
46/// `T` must be [`Ord`] to ensure comparability.
47#[derive(Clone, Debug)]
48pub struct DBF {
49    n: usize,
50    dbf: Vec<Vec<usize>>,
51}
52
53impl DBF {
54    /// Constructs a DBF from a slice of `T`
55    ///
56    /// Complexity: $\mathcal{O}(N\log N)$ where
57    /// - $N$ is the size of the slice.
58    ///
59    /// # Examples
60    ///
61    /// ```
62    /// use cp_library::string::DBF;
63    /// let dbf = DBF::from("banana".as_bytes());
64    /// ```
65    pub fn from<T: Ord>(arr: &[T]) -> Self {
66        let n = arr.len();
67
68        let mut order = (0..n).collect::<Vec<_>>();
69        order.sort_unstable_by_key(|i| &arr[*i]);
70
71        let mut row = vec![0; n];
72        row[order[0]] = 0;
73
74        for i in 1..n {
75            row[order[i]] = row[order[i - 1]] + ((arr[order[i]] != arr[order[i - 1]]) as usize);
76        }
77
78        let mut freq = vec![0; n];
79        let mut dbf = vec![row.clone()];
80
81        while {
82            let last_row = &dbf[dbf.len() - 1];
83
84            freq.fill(0);
85
86            for v in last_row {
87                freq[*v] += 1;
88            }
89            let mut acc = 0;
90
91            #[allow(clippy::needless_range_loop)]
92            for i in 0..n {
93                acc += freq[i];
94                freq[i] = acc - freq[i];
95            }
96
97            let off = 1 << (dbf.len() - 1);
98            for i in 0..n {
99                let elem = last_row[(i + off) % n];
100                row[freq[elem]] = i;
101                freq[elem] += 1;
102            }
103
104            freq.fill(0);
105            for v in last_row {
106                freq[*v] += 1;
107            }
108
109            let mut acc = 0;
110
111            #[allow(clippy::needless_range_loop)]
112            for i in 0..n {
113                acc += freq[i];
114                freq[i] = acc - freq[i];
115            }
116
117            for i in &row {
118                let elem = last_row[*i];
119                order[freq[elem]] = *i;
120                freq[elem] += 1;
121            }
122
123            row[order[0]] = 0;
124
125            for i in 1..n {
126                let fst_pair = (last_row[order[i]], last_row[(order[i] + off) % n]);
127                let snd_pair = (last_row[order[i - 1]], last_row[(order[i - 1] + off) % n]);
128
129                row[order[i]] = row[order[i - 1]] + ((fst_pair != snd_pair) as usize);
130            }
131
132            dbf.push(row.clone());
133
134            (1 << (dbf.len() - 1)) < n
135        } {}
136
137        Self { n, dbf }
138    }
139
140    /// Returns the last row of the DBF (order of cyclic shifts)
141    ///
142    /// Complexity; $\mathcal{O}(N)$ where:
143    /// - $N$ is the size of DBF
144    ///
145    /// The complexity is due to the copy of the last row
146    ///
147    /// # Examples
148    ///
149    /// ```
150    /// use cp_library::string::DBF;
151    /// let dbf = DBF::from("banana".as_bytes());
152    /// let last_row = dbf.last_row();
153    ///
154    /// assert_eq!(last_row, vec![3, 2, 5, 1, 4, 0]);
155    /// ```
156    pub fn last_row(&self) -> Vec<usize> {
157        self.dbf[self.dbf.len() - 1].clone()
158    }
159
160    /// Gives a representative of a substring $[l, r)$ of the slice.
161    ///
162    /// Complexity: $\mathcal{O}(1)$
163    ///
164    /// Returns a pair of [`usize`] representing the substring [l, r).
165    /// Substrings of the same length can be compared by comparing
166    /// their representatives.
167    ///
168    /// # Example
169    ///
170    /// ```
171    /// use cp_library::string::DBF;
172    /// let dbf = DBF::from("banana".as_bytes());
173    ///
174    /// assert_eq!(dbf.get_repr(1, 4), (1, 3));
175    /// ```
176    ///
177    /// # Panics
178    ///
179    /// Only in debug builds, if [l, r) does not specify a valid range
180    /// ```should_panic
181    /// use cp_library::string::DBF;
182    /// let dbf = DBF::from("banana".as_bytes());
183    /// let repr = dbf.get_repr(5, 8);
184    /// ```
185    pub fn get_repr(&self, l: usize, r: usize) -> (usize, usize) {
186        debug_assert!(l < r && r <= self.n);
187
188        let h = 63 - (r - l).leading_zeros();
189        (
190            self.dbf[h as usize][l],
191            self.dbf[h as usize][(r - (1 << h)) % self.n],
192        )
193    }
194
195    /// Compares two substrings $[l_1, r_1)$ and $[l_2, r_2)$
196    ///
197    /// Complexity: $\mathcal{O}(1)$
198    ///
199    /// # Examples
200    ///
201    /// ```
202    /// use std::cmp::Ordering;
203    /// use cp_library::string::DBF;
204    /// let dbf = DBF::from("banana".as_bytes());
205    ///
206    /// assert_eq!(dbf.cmp(1, 4, 0, 2), Ordering::Less)
207    /// ```
208    ///
209    /// # Panics
210    ///
211    /// Only in debug builds, if either [l1, r1) or [l2, r2) do not specify valid range
212    ///
213    /// ```should_panic
214    /// use cp_library::string::DBF;
215    /// let dbf = DBF::from("banana".as_bytes());
216    /// let ord = dbf.cmp(3, 1, 0, 3);
217    /// ```
218    pub fn cmp(&self, l1: usize, r1: usize, l2: usize, r2: usize) -> cmp::Ordering {
219        debug_assert!(l1 < r1 && r1 <= self.n);
220        debug_assert!(l2 < r2 && r2 <= self.n);
221
222        let common_length = cmp::min(r1 - l1, r2 - l2);
223        let repr1 = self.get_repr(l1, l1 + common_length);
224        let repr2 = self.get_repr(l2, l2 + common_length);
225
226        match repr1.cmp(&repr2) {
227            cmp::Ordering::Equal => (r1 - l1).cmp(&(r2 - l2)),
228            order => order,
229        }
230    }
231}

233/// Constructs the [Suffix array](https://cp-algorithms.com/string/suffix-array.html) of a slice of `T`
234///
235/// `T` must be [`Ord`]
236///
237/// Complexity: $\mathcal{O}(N\log N)$ comparisons where:
238/// - $N$ is the size of the slice.
239///
240/// # Examples
241///
242/// ```
243/// use cp_library::string::suffix_array;
244/// let suff_arr = suffix_array("banana".as_bytes());
245///
246/// assert_eq!(suff_arr, [5, 3, 1, 0, 4, 2]);
247/// ```
248pub fn suffix_array<T: Ord>(arr: &[T]) -> Vec<usize> {
249    let n = arr.len();
250    let arr = arr
251        .iter()
252        .map(Some)
253        .chain(std::iter::once(None))
254        .collect::<Vec<_>>();
255    let dbf = DBF::from(&arr);
256    let last_row = dbf.last_row();
257
258    let mut suffix_array = vec![0; n];
259    for (i, &x) in last_row.iter().enumerate() {
260        if x != 0 {
261            suffix_array[x - 1] = i;
262        }
263    }
264
265    suffix_array
266}

268/// Constructs the [Suffix array](https://cp-algorithms.com/string/suffix-array.html)
269/// and [LCP](https://cp-algorithms.com/string/suffix-array.html#longest-common-prefix-of-two-substrings-without-additional-memory)
270/// array of a slice of `T`
271///
272/// Complexity: $\mathcal{O}(N\log N)$ comparisons where:
273/// - $N$ is the size of the slice.
274///
275/// # Examples
276///
277/// ```
278/// use cp_library::string::lcp;
279/// let (suff_arr, lcp_arr) = lcp("banana".as_bytes());
280///
281/// assert_eq!(suff_arr, [5, 3, 1, 0, 4, 2]);
282/// assert_eq!(lcp_arr, [0, 1, 3, 0, 0, 2]);
283/// ```
284pub fn lcp<T: Ord>(arr: &[T]) -> (Vec<usize>, Vec<usize>) {
285    let n = arr.len();
286    let suffix_array = suffix_array(arr);
287
288    let mut inv = vec![0; n];
289    for (i, &x) in suffix_array.iter().enumerate() {
290        inv[x] = i;
291    }
292
293    let mut lcp = vec![0; n];
294    let mut k = 0;
295
296    for i in 0..n {
297        if inv[i] == n - 1 {
298            k = 0;
299        } else {
300            let j = suffix_array[inv[i] + 1];
301            while i + k < n && j + k < n && arr[i + k] == arr[j + k] {
302                k += 1;
303            }
304
305            lcp[inv[i] + 1] = k;
306            k = k.saturating_sub(1);
307        }
308    }
309
310    (suffix_array, lcp)
